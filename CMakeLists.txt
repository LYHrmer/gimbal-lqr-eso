cmake_minimum_required(VERSION 3.16)
project(robomaster_gimbal_lqr_eso VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(YAW_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)

add_library(yaw_controller SHARED src/yaw_controller.c src/dm_mit.c src/gm6020.c)
add_library(gimbal_dm4310 STATIC src/yaw_controller.c src/dm_mit.c)
add_library(gimbal_gm6020 STATIC src/yaw_controller.c src/gm6020.c)
foreach(controller_target IN ITEMS yaw_controller gimbal_dm4310 gimbal_gm6020)
    target_include_directories(${controller_target} PUBLIC include)
    target_link_libraries(${controller_target} PUBLIC m)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${controller_target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wshadow)
        if(YAW_SANITIZERS)
            target_compile_options(${controller_target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${controller_target} PUBLIC -fsanitize=address,undefined)
        endif()
    endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
    foreach(test_name IN ITEMS yaw_controller dm_mit gm6020)
        add_executable(test_${test_name} tests/test_${test_name}.c)
        target_link_libraries(test_${test_name} PRIVATE yaw_controller)
        if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(test_${test_name} PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
        endif()
        add_test(NAME ${test_name} COMMAND test_${test_name})
    endforeach()
    add_executable(test_stm32_yaw_periodic tests/test_stm32_yaw_periodic.c
                   examples/stm32/yaw_periodic.c)
    target_include_directories(test_stm32_yaw_periodic PRIVATE examples/stm32)
    target_link_libraries(test_stm32_yaw_periodic PRIVATE yaw_controller)
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(test_stm32_yaw_periodic PRIVATE -Wall -Wextra -Wpedantic
                               -Wconversion -Wshadow -Werror -UNDEBUG)
    endif()
    add_test(NAME stm32_yaw_periodic COMMAND test_stm32_yaw_periodic)
endif()
