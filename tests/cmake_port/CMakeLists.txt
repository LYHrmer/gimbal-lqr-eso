cmake_minimum_required(VERSION 3.16)
project(gimbal_port_consumer LANGUAGES C CXX)

set(PORT_TEST_MOTOR "GM6020" CACHE STRING "Motor codec to link")
set(PORT_TEST_LINK_STYLE "NONE" CACHE STRING "Host link signature: NONE, PLAIN or KEYWORD")
option(PORT_TEST_LINK_BEFORE "Host links a dependency before attaching gimbal" ON)
add_executable(port_consumer main.cpp)
add_library(port_host_dependency INTERFACE)
function(port_test_link_host_dependency)
    if(PORT_TEST_LINK_STYLE STREQUAL "PLAIN")
        target_link_libraries(port_consumer port_host_dependency)
    elseif(PORT_TEST_LINK_STYLE STREQUAL "KEYWORD")
        target_link_libraries(port_consumer PRIVATE port_host_dependency)
    endif()
endfunction()
if(PORT_TEST_LINK_BEFORE)
    port_test_link_host_dependency()
endif()
target_compile_features(port_consumer PRIVATE cxx_std_17)
target_compile_definitions(port_consumer PRIVATE PORT_MOTOR_${PORT_TEST_MOTOR}=1)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(port_consumer PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/GimbalController.cmake")
gimbal_add_to_target(port_consumer ${PORT_TEST_MOTOR})
# A helper must not force the signature of later host link calls either.
port_test_link_host_dependency()
get_target_property(port_links port_consumer LINK_LIBRARIES)
if(NOT "m" IN_LIST port_links)
    message(FATAL_ERROR "gimbal_add_to_target() must retain the math dependency")
endif()
if(PORT_TEST_DUPLICATE)
    gimbal_add_to_target(port_consumer ${PORT_TEST_MOTOR})
endif()
